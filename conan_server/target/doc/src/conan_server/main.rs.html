28// ─── Shared State ──────────────────────────────────────────────────────────────

30struct MatchManager {
31    catalog_path: PathBuf,
32    deck_path: PathBuf,
33    catalog: CardCatalog,
34    deck_def: DeckDefinition,
35    engine: MatchEngine,
36    p2_is_cpu: bool,
37    cpu_agent: Option<RandomAgent>,
38}
39
40impl MatchManager {
41    fn new() -> Result<Self, Box<dyn std::error::Error>> {
42        let catalog_path = PathBuf::from("data/dummy/cards.json");
43        let deck_path = PathBuf::from("data/dummy/deck.json");
44
45        let catalog = CardCatalog::load_from_json(&catalog_path)?;
46        let deck_def = Self::load_deck(&deck_path)?;
47
48        let mut mgr = MatchManager {
49            catalog_path,
50            deck_path,
51            catalog: catalog.clone(),
52            deck_def: deck_def.clone(),
53            engine: MatchEngine::new(catalog),
54            p2_is_cpu: true,
55            cpu_agent: Some(RandomAgent::new("player2")),
56        };
57
58        mgr.reset_match_inner(&deck_def);
59        Ok(mgr)
60    }
61
62    fn load_deck(path: &PathBuf) -> Result<DeckDefinition, Box<dyn std::error::Error>> {
63        let content = std::fs::read_to_string(path)?;
64        let deck: DeckDefinition = serde_json::from_str(&content)?;
65        Ok(deck)
66    }
67
68    fn reset_match_inner(&mut self, deck_def: &DeckDefinition) {
69        let mut engine = MatchEngine::new(self.catalog.clone());
70        engine.add_player("player1", deck_def);
71        engine.add_player("player2", deck_def);
72        engine.start_match();
73        self.engine = engine;
74        self.process_cpu_if_needed();
75    }
76
77    fn reset_match(&mut self) {
78        let deck_def = match Self::load_deck(&self.deck_path) {
79            Ok(d) => d,
80            Err(_) => self.deck_def.clone(),
81        };
82        self.deck_def = deck_def.clone();
83        self.reset_match_inner(&deck_def);
84    }
85
86    fn save_deck(&mut self, new_def: DeckDefinition) -> Result<(), String> {
87        let content = serde_json::to_string_pretty(&new_def)
88            .map_err(|e| e.to_string())?;
89        std::fs::write(&self.deck_path, content).map_err(|e| e.to_string())?;
90        self.deck_def = new_def;
91        Ok(())
92    }
93
94    fn process_cpu_if_needed(&mut self) {
95        let max_steps = 50;
96        let mut steps = 0;
97        loop {
98            if steps >= max_steps || !self.p2_is_cpu {
99                break;
100            }
101            match &self.engine.pending_decision {
102                Some(dec) => {
103                    if dec.get("playerId").and_then(|v| v.as_str()) != Some("player2") {
104                        break;
105                    }
106                    let ans = match &self.cpu_agent {
107                        Some(agent) => agent.choose_action(dec),
108                        None => break,
109                    };
110                    if self.engine.submit_decision("player2", &ans).is_err() {
111                        break;
112                    }
113                }
114                None => break,
115            }
116            steps += 1;
117        }
118    }
119}

148// ─── Handlers ─────────────────────────────────────────────────────────────────

150async fn handle_state(
151    State(state): State<SharedState>,
152    Query(params): Query<PlayerIdQuery>,
153) -> impl IntoResponse {
154    let mut mgr = state.lock().unwrap();
155    mgr.process_cpu_if_needed();
156
157    let player_id = params
158        .player_id
159        .filter(|id| mgr.engine.players.contains_key(id.as_str()))
160        .unwrap_or_else(|| "player1".to_string());
161
162    let mut obs = build_player_observation(&mgr.engine, &player_id);
163    obs["p2IsCpu"] = json!(mgr.p2_is_cpu);
164
165    Json(obs).into_response()
166}
167
168async fn handle_catalog(State(state): State<SharedState>) -> impl IntoResponse {
169    let mgr = state.lock().unwrap();
170    let total = mgr.catalog.definitions.len();
171    let cats = mgr.catalog.raw_json.clone();
172
173    // Build normalized response
174    let mut resp = cats.clone();
175    resp["totalCards"] = json!(total);
176
177    Json(resp).into_response()
178}
179
180async fn handle_deck(State(state): State<SharedState>) -> impl IntoResponse {
181    let mgr = state.lock().unwrap();
182    match std::fs::read_to_string(&mgr.deck_path) {
183        Ok(content) => {
184            let val: Value = serde_json::from_str(&content).unwrap_or(json!({}));
185            Json(val).into_response()
186        }
187        Err(e) => (
188            StatusCode::INTERNAL_SERVER_ERROR,
189            Json(json!({"error": e.to_string()})),
190        )
191            .into_response(),
192    }
193}
194
195async fn handle_reset(State(state): State<SharedState>) -> impl IntoResponse {
196    let mut mgr = state.lock().unwrap();
197    mgr.reset_match();
198    Json(StatusOk { status: "ok" }).into_response()
199}
200
201async fn handle_set_cpu(
202    State(state): State<SharedState>,
203    Json(body): Json<SetCpuRequest>,
204) -> impl IntoResponse {
205    let mut mgr = state.lock().unwrap();
206    mgr.p2_is_cpu = body.is_cpu;
207    mgr.cpu_agent = if body.is_cpu {
208        Some(RandomAgent::new("player2"))
209    } else {
210        None
211    };
212    mgr.process_cpu_if_needed();
213    Json(StatusOk { status: "ok" }).into_response()
214}
215
216async fn handle_save_deck(
217    State(state): State<SharedState>,
218    Json(body): Json<DeckDefinition>,
219) -> impl IntoResponse {
220    // Validate
221    let total: i32 = body.main_deck.iter().map(|item| item.count).sum();
222    if total != 40 {
223        return (
224            StatusCode::BAD_REQUEST,
225            Json(json!({"error": format!("メインデッキはちょうど40枚である必要があります（現在: {}枚）", total)})),
226        )
227            .into_response();
228    }
229    for item in &body.main_deck {
230        if item.count > 3 {
231            return (
232                StatusCode::BAD_REQUEST,
233                Json(json!({"error": format!("同一カードは最大3枚までです: {}", item.definition_id)})),
234            )
235                .into_response();
236        }
237    }
238
239    let mut mgr = state.lock().unwrap();
240    if let Err(e) = mgr.save_deck(body) {
241        return (
242            StatusCode::BAD_REQUEST,
243            Json(json!({"error": e})),
244        )
245            .into_response();
246    }
247    mgr.reset_match();
248    Json(StatusOk { status: "ok" }).into_response()
249}
250
251async fn handle_decision(
252    State(state): State<SharedState>,
253    Json(body): Json<DecisionRequest>,
254) -> impl IntoResponse {
255    let mut mgr = state.lock().unwrap();
256    let ans = json!({ "value": body.value.unwrap_or(json!({})) });
257    match mgr.engine.submit_decision(&body.player_id, &ans) {
258        Ok(_) => {
259            mgr.process_cpu_if_needed();
260            Json(StatusOk { status: "ok" }).into_response()
261        }
262        Err(e) => (
263            StatusCode::BAD_REQUEST,
264            Json(json!({"error": e})),
265        )
266            .into_response(),
267    }
268}

272#[tokio::main]
273async fn main() -> Result<(), Box<dyn std::error::Error>> {
274    tracing_subscriber::fmt::init();
275
276    let mgr = MatchManager::new().expect("MatchManager 初期化に失敗しました");
277    let shared_state: SharedState = Arc::new(Mutex::new(mgr));
278
279    // Static files directory (card images, HTML)
280    let static_dir = PathBuf::from("src/conan_engine/web/static");
281
282    let app = Router::new()
283        .route("/api/state", get(handle_state))
284        .route("/api/catalog", get(handle_catalog))
285        .route("/api/deck", get(handle_deck))
286        .route("/api/reset", post(handle_reset))
287        .route("/api/set_cpu", post(handle_set_cpu))
288        .route("/api/save_deck", post(handle_save_deck))
289        .route("/api/decision", post(handle_decision))
290        .fallback_service(ServeDir::new(&static_dir))
291        .with_state(shared_state);
292
293    let port: u16 = std::env::args()
294        .nth(1)
295        .and_then(|s| s.parse().ok())
296        .unwrap_or(8000);
297
298    let addr = SocketAddr::from(([0, 0, 0, 0], port));
299    println!("🚀 Detective Conan TCG Server (Rust) running at http://localhost:{port}");
300
301    let listener = tokio::net::TcpListener::bind(addr).await?;
302    axum::serve(listener, app).await?;
303
304    Ok(())
305}
