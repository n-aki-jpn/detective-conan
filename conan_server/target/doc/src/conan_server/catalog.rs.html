1use crate::models::{CardDefinition, CardType};
2use serde_json::Value;
3use std::collections::HashMap;
4use std::fs::File;
5use std::io::BufReader;
6use std::path::Path;
7
8#[derive(Debug, Clone)]
9pub struct CardCatalog {
10    pub catalog_version: String,
11    pub definitions: HashMap<String, CardDefinition>,
12    pub raw_json: Value,
13}
14
15impl CardCatalog {
16    pub fn load_from_json<P: AsRef<Path>>(file_path: P) -> Result<Self, Box<dyn std::error::Error>> {
17        let file = File::open(file_path)?;
18        let reader = BufReader::new(file);
19        let mut raw_json: Value = serde_json::from_reader(reader)?;
20
21        let catalog_version = raw_json
22            .get("catalogVersion")
23            .and_then(|v| v.as_str())
24            .unwrap_or("official-raw-v1")
25            .to_string();
26
27        let mut definitions = HashMap::new();
28
29        if let Some(cards_val) = raw_json.get_mut("cards").and_then(|v| v.as_array_mut()) {
30            for card in cards_val.iter_mut() {
31                let def_id = match card.get("definitionId").and_then(|v| v.as_str()) {
32                    Some(id) => id.to_string(),
33                    None => continue,
34                };
35
36                let raw_type_str = card
37                    .get("cardType")
38                    .or_else(|| card.get("type"))
39                    .and_then(|v| v.as_str())
40                    .unwrap_or("CHARACTER");
41
42                let card_type = match raw_type_str.trim() {
43                    "パートナー" | "PARTNER" => CardType::PARTNER,
44                    "事件" | "CASE" => CardType::CASE,
45                    "イベント" | "EVENT" => CardType::EVENT,
46                    _ => CardType::CHARACTER,
47                };
48
49                // Also update the card object in raw_json so that /api/catalog serves normalized types
50                let normalized_type_str = match card_type {
51                    CardType::PARTNER => "PARTNER",
52                    CardType::CASE => "CASE",
53                    CardType::EVENT => "EVENT",
54                    CardType::CHARACTER => "CHARACTER",
55                };
56                card["cardType"] = Value::String(normalized_type_str.to_string());
57
58                let canonical_id = card
59                    .get("canonicalId")
60                    .and_then(|v| v.as_str())
61                    .unwrap_or(&def_id)
62                    .to_string();
63
64                let colors = card
65                    .get("colors")
66                    .and_then(|v| v.as_array())
67                    .map(|arr| {
68                        arr.iter()
69                            .filter_map(|c| c.as_str().map(|s| s.to_string()))
70                            .collect()
71                    })
72                    .unwrap_or_default();
73
74                let parse_i32 = |v: Option<&Value>| -> i32 {
75                    match v {
76                        Some(Value::Number(n)) => n.as_i64().unwrap_or(0) as i32,
77                        Some(Value::String(s)) => s.parse::<i32>().unwrap_or(0),
78                        _ => 0,
79                    }
80                };
81
82                let level = parse_i32(card.get("level"));
83                let ap = parse_i32(card.get("ap"));
84                let lp = parse_i32(card.get("lp"));
85
86                let name = card
87                    .get("title")
88                    .or_else(|| card.get("dummyName"))
89                    .and_then(|v| v.as_str())
90                    .unwrap_or(&def_id)
91                    .to_string();
92
93                let names = card
94                    .get("names")
95                    .and_then(|v| v.as_array())
96                    .map(|arr| {
97                        arr.iter()
98                            .filter_map(|c| c.as_str().map(|s| s.to_string()))
99                            .collect()
100                    })
101                    .unwrap_or_else(|| vec![name]);
102
103                let description = card
104                    .get("description")
105                    .and_then(|v| v.as_str())
106                    .unwrap_or("")
107                    .to_string();
108
109                let main_thumb = card
110                    .get("mainThumb")
111                    .and_then(|v| v.as_str())
112                    .map(|s| s.to_string());
113                let main_path = card
114                    .get("mainPath")
115                    .and_then(|v| v.as_str())
116                    .map(|s| s.to_string());
117
118                let case_levels = card.get("caseLevels").and_then(|v| {
119                    if let Some(obj) = v.as_object() {
120                        let mut map = HashMap::new();
121                        for (k, val) in obj {
122                            map.insert(k.clone(), parse_i32(Some(val)));
123                        }
124                        Some(map)
125                    } else {
126                        None
127                    }
128                });
129
130                let c_def = CardDefinition {
131                    definition_id: def_id.clone(),
132                    canonical_id,
133                    card_type,
134                    colors,
135                    level,
136                    ap,
137                    lp,
138                    names,
139                    description,
140                    traits: vec![],
141                    case_levels,
142                    main_thumb,
143                    main_path,
144                };
145
146                definitions.insert(def_id, c_def);
147            }
148        }
149
150        Ok(CardCatalog {
151            catalog_version,
152            definitions,
153            raw_json,
154        })
155    }
156
157    pub fn get(&self, definition_id: &str) -> Option<&CardDefinition> {
158        self.definitions.get(definition_id)
159    }
160}
