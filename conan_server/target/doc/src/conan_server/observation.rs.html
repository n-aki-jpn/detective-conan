1use crate::engine::{MatchEngine, PlayerState};
2use crate::models::{CardInstance, CardOrientation, CasePhase, MatchPhase};
3use serde_json::{json, Value};
4
5pub fn build_player_observation(engine: &MatchEngine, viewing_player_id: &str) -> Value {
6    let catalog = &engine.catalog;
7
8    let serialize_card = |c: &CardInstance, is_secret: bool| -> Value {
9        let orientation_str = match c.orientation {
10            CardOrientation::ACTIVE => "ACTIVE",
11            CardOrientation::SLEEP => "SLEEP",
12            CardOrientation::STUN => "STUN",
13        };
14
15        if is_secret {
16            return json!({
17                "instanceId": c.instance_id,
18                "faceUp": false,
19                "orientation": orientation_str
20            });
21        }
22
23        let c_def = catalog.get(&c.definition_id);
24        let name = c_def.and_then(|d| d.names.first().cloned()).unwrap_or_else(|| c.definition_id.clone());
25        let desc = c_def.map(|d| d.description.clone()).unwrap_or_default();
26        let colors = c_def.map(|d| d.colors.clone()).unwrap_or_default();
27        let level = c_def.map(|d| d.level).unwrap_or(0);
28        let ap = c_def.map(|d| d.ap).unwrap_or(0);
29        let lp = c_def.map(|d| d.lp).unwrap_or(0);
30        let card_type = c_def.map(|d| format!("{:?}", d.card_type)).unwrap_or_else(|| "CHARACTER".to_string());
31        let main_thumb = c_def.and_then(|d| d.main_thumb.clone());
32        let main_path = c_def.and_then(|d| d.main_path.clone());
33
34        json!({
35            "instanceId": c.instance_id,
36            "definitionId": c.definition_id,
37            "name": name,
38            "description": desc,
39            "colors": colors,
40            "level": level,
41            "ap": ap,
42            "lp": lp,
43            "cardType": card_type,
44            "orientation": orientation_str,
45            "enteredTurn": c.entered_turn,
46            "faceUp": true,
47            "mainThumb": main_thumb,
48            "mainPath": main_path
49        })
50    };
51
52    let serialize_player = |ps: &PlayerState, is_me: bool| -> Value {
53        let case_phase_str = match ps.case_phase {
54            CasePhase::INTRO => "INTRO",
55            CasePhase::RESOLUTION => "RESOLUTION",
56        };
57
58        json!({
59            "playerId": ps.player_id,
60            "isFirstPlayer": ps.is_first_player,
61            "casePhase": case_phase_str,
62            "fileCount": ps.effective_file_count(),
63            "deckCount": ps.deck.len(),
64            "evidenceCount": ps.evidence.len(),
65            "removeCount": ps.remove.len(),
66            "handCount": ps.hand.len(),
67            "hand": ps.hand.iter().map(|c| serialize_card(c, !is_me)).collect::<Vec<_>>(),
68            "field": ps.field.iter().map(|c| serialize_card(c, false)).collect::<Vec<_>>(),
69            "partner": ps.partner.as_ref().map(|c| serialize_card(c, false)),
70            "case": ps.r#case.as_ref().map(|c| serialize_card(c, false)),
71            "remove": ps.remove.iter().map(|c| serialize_card(c, false)).collect::<Vec<_>>(),
72            "normalHandPlayed": ps.normal_hand_played_this_turn,
73            "hintPlayed": ps.hint_played_this_turn
74        })
75    };
76
77    let me_ps = engine.players.get(viewing_player_id);
78    let opp_pid = engine.player_order.iter().find(|pid| *pid != viewing_player_id).cloned().unwrap_or_default();
79    let opp_ps = engine.players.get(&opp_pid);
80
81    let pending_dec = match &engine.pending_decision {
82        Some(dec) if dec.get("playerId").and_then(|v| v.as_str()) == Some(viewing_player_id) => {
83            Some(dec.clone())
84        }
85        _ => None,
86    };
87
88    let decision_player_id = engine.pending_decision.as_ref().and_then(|d| d.get("playerId").and_then(|v| v.as_str()));
89
90    let phase_str = match engine.phase {
91        MatchPhase::WAITING => "WAITING",
92        MatchPhase::SETUP_HIDDEN => "SETUP_HIDDEN",
93        MatchPhase::MULLIGAN_FIRST => "MULLIGAN_FIRST",
94        MatchPhase::MULLIGAN_SECOND => "MULLIGAN_SECOND",
95        MatchPhase::REVEAL => "REVEAL",
96        MatchPhase::AUTO_RESTORE_PARTNER => "AUTO_RESTORE_PARTNER",
97        MatchPhase::AUTO_ACTIVATE => "AUTO_ACTIVATE",
98        MatchPhase::AUTO_DRAW => "AUTO_DRAW",
99        MatchPhase::AUTO_FILE => "AUTO_FILE",
100        MatchPhase::MAIN => "MAIN",
101        MatchPhase::DEDUCE => "DEDUCE",
102        MatchPhase::ACTION_GUARD => "ACTION_GUARD",
103        MatchPhase::ACTION_CONTACT => "ACTION_CONTACT",
104        MatchPhase::ACTION_CASE => "ACTION_CASE",
105        MatchPhase::END_TRIGGERS => "END_TRIGGERS",
106        MatchPhase::END_EXPIRE => "END_EXPIRE",
107        MatchPhase::FINISHED => "FINISHED",
108    };
109
110    json!({
111        "matchId": "match_live",
112        "revision": engine.revision,
113        "phase": phase_str,
114        "turn": engine.turn,
115        "turnPlayerId": engine.turn_player_id,
116        "decisionPlayerId": decision_player_id,
117        "viewingPlayerId": viewing_player_id,
118        "me": me_ps.map(|p| serialize_player(p, true)).unwrap_or(Value::Null),
119        "opponent": opp_ps.map(|p| serialize_player(p, false)).unwrap_or(Value::Null),
120        "pendingDecision": pending_dec,
121        "winnerId": engine.winner_id,
122        "finishReason": engine.finish_reason
123    })
124}
